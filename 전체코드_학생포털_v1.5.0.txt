function fmtDate(s){
  if(!s)return "-";
  const d=new Date(`${s}T12:00:00`);
  if(Number.isNaN(d.getTime()))return s;
  return `${d.getMonth()+1}월 ${d.getDate()}일`;
}
function fmtDateTime(v){
  if(!v)return "-";
  try{
    const d=v?.toDate?v.toDate():new Date(v);
    if(Number.isNaN(d.getTime()))return "-";
    return `${d.getFullYear()}.${d.getMonth()+1}.${d.getDate()} ${String(d.getHours()).padStart(2,"0")}:${String(d.getMinutes()).padStart(2,"0")}`;
  }catch{return "-";}
}
function fmtMaterialDate(v){
  if(!v)return "-";
  const d=new Date(v);
  if(Number.isNaN(d.getTime()))return String(v).slice(0,10)||"-";
  return `${d.getFullYear()}.${String(d.getMonth()+1).padStart(2,"0")}.${String(d.getDate()).padStart(2,"0")}`;
}

function formatNoticeBytes(bytes){const n=Number(bytes||0);if(n<1024)return `${n}B`;if(n<1024*1024)return `${(n/1024).toFixed(1)}KB`;return `${(n/1024/1024).toFixed(1)}MB`;}

function normalizeNameKey(v){return String(v||"").replace(/\s+/g,"").trim();}
function openAssessments(){
  if(adminPreviewData?.assessment)return [{...adminPreviewData.assessment,status:"OPEN"}];
  return publicAssessments.filter(a=>a.status==="OPEN").sort((a,b)=>String(b.updatedAt?.seconds||b.updatedAt||"").localeCompare(String(a.updatedAt?.seconds||a.updatedAt||"")));

function assessmentById(id){
  if(adminPreviewData?.assessment&&String(adminPreviewData.assessment.id)===String(id))return {...adminPreviewData.assessment,status:"OPEN"};
  return publicAssessments.find(a=>a.id===id)||null;

function applyAdminPreviewPayload(data){
  if(!adminPreviewMode||!data?.assessment||!data?.student)return;
  if(previewRunParam&&String(data.previewRun||"")!==String(previewRunParam))return;
  if(adminPreviewData&&String(adminPreviewData.previewRun||"")===String(data.previewRun||""))return;
  adminPreviewData=data;
  $("adminPreviewBanner")?.classList.remove("hidden");
  selectedClass=String(data.student.className||selectedClass||"");
  selectedAssessmentId=null;activeAttempt=null;assessmentContent=null;assessmentAnswers={};assessmentDirty=false;
  setView("assessment");renderClassTabs();
  const start=data.startAt||"list";
  if(start==="login"){selectedAssessmentId=data.assessment.id;}
  if(start==="exam"){
    selectedAssessmentId=data.assessment.id;
    assessmentContent={assessmentId:data.assessment.id,title:data.assessment.title,instructions:data.assessment.instructions||"",questions:data.assessment.questions||[]};
    activeAttempt={studentId:String(data.student.studentId),studentName:String(data.student.name),status:"IN_PROGRESS",preview:true};
  }
  renderAssessmentPortal();
}
function adminPreviewHost(){
  if(window.opener&&!window.opener.closed)return window.opener;
  if(window.parent&&window.parent!==window)return window.parent;
  return null;
}
function notifyAdminPreviewReady(){
  const host=adminPreviewHost();
  if(!host)return;
  try{host.postMessage({type:"JCOUP_PREVIEW_READY"},location.origin);}catch{}
}
window.addEventListener("message",e=>{
  if(!adminPreviewMode||e.origin!==location.origin)return;
  const host=adminPreviewHost();
  if(host&&e.source!==host)return;
  if(e.data?.type==="JCOUP_ADMIN_PREVIEW")applyAdminPreviewPayload(e.data.payload);
});

async function ensureAssessmentAuth(){
  if(!fb)throw new Error("Firebase 연결이 아직 준비되지 않았습니다.");
  if(fb.auth.currentUser){authUser=fb.auth.currentUser;return authUser;}
  const cred=await fb.authMod.signInAnonymously(fb.auth);authUser=cred.user;return authUser;
}
async function enterAssessment(e){
  e.preventDefault();const msg=$("assessmentLoginMessage"),btn=e.submitter;msg.textContent="인증 중...";btn.disabled=true;
  const studentId=$("assessmentStudentId").value.trim(),studentName=$("assessmentStudentName").value.trim(),code=$("assessmentCode").value.trim();
  if(!/^\d{4,5}$/.test(studentId)){msg.textContent="학번은 4자리 또는 5자리 숫자로 입력하세요. 예: 20101";btn.disabled=false;return;}
  try{
    const a=assessmentById(selectedAssessmentId);if(!a||a.status!=="OPEN")throw new Error("현재 응시 가능한 평가가 아닙니다.");
    if(adminPreviewMode&&adminPreviewData){
      const ps=adminPreviewData.student||{};
      if(String(studentId)!==String(ps.studentId)||normalizeNameKey(studentName)!==normalizeNameKey(ps.name)||String(code)!==String(a.accessCode||""))throw new Error("테스트 정보가 일치하지 않습니다.");
      assessmentContent={assessmentId:a.id,title:a.title,instructions:a.instructions||"",questions:a.questions||[]};
      activeAttempt={studentId,studentName,status:"IN_PROGRESS",preview:true};assessmentAnswers={};assessmentDirty=false;msg.textContent="";renderAssessmentPortal();return;
    }
    const user=await ensureAssessmentAuth();
    const sessionRef=fb.fsMod.doc(fb.db,"publicCourses",projectId,"assessments",a.id,"sessions",user.uid);
    await fb.fsMod.setDoc(sessionRef,{authUid:user.uid,studentId,studentName,nameKey:normalizeNameKey(studentName),accessCodeProof:code,lastSeenAt:fb.fsMod.serverTimestamp(),createdAt:fb.fsMod.serverTimestamp()},{merge:true});
    const contentRef=fb.fsMod.doc(fb.db,"publicCourses",projectId,"assessments",a.id,"content","main"),attemptRef=fb.fsMod.doc(fb.db,"publicCourses",projectId,"assessments",a.id,"attempts",studentId);
    const [contentSnap,attemptSnap]=await Promise.all([fb.fsMod.getDoc(contentRef),fb.fsMod.getDoc(attemptRef)]);if(!contentSnap.exists())throw new Error("평가 문항을 불러올 수 없습니다.");
    assessmentContent={id:contentSnap.id,...contentSnap.data()};
    if(attemptSnap.exists()){activeAttempt={id:attemptSnap.id,...attemptSnap.data(),studentId};assessmentAnswers={...(activeAttempt.answers||{})};}
    else{let backup={};try{backup=JSON.parse(localStorage.getItem(assessmentBackupKey(a.id,studentId))||"{}")||{};}catch{}assessmentAnswers=backup;await fb.fsMod.setDoc(attemptRef,{studentId,studentName,authUid:user.uid,status:"IN_PROGRESS",answers:assessmentAnswers,currentQuestion:1,startedAt:fb.fsMod.serverTimestamp(),lastSeenAt:fb.fsMod.serverTimestamp(),lastSavedAt:fb.fsMod.serverTimestamp()});const fresh=await fb.fsMod.getDoc(attemptRef);activeAttempt={id:fresh.id,...fresh.data(),studentId};}
    assessmentDirty=false;renderAssessmentPortal();
  }catch(err){console.error(err);msg.textContent="입장할 수 없습니다. 학번·이름·응시코드를 다시 확인하세요.";}finally{btn.disabled=false;}
}

async function saveAssessmentAttempt({submit=false,heartbeat=false}={}){
  if(!activeAttempt||!selectedAssessmentId||assessmentSaving||activeAttempt.status==="SUBMITTED")return;
  assessmentSaving=true;const st=$("assessmentSaveState");if(st&&!heartbeat)st.textContent=adminPreviewMode?"테스트 저장 중...":"저장 중...";
  try{
    if(adminPreviewMode&&adminPreviewData){
      assessmentDirty=false;
      if(st)st.textContent=submit?"테스트 제출 완료 · 서버 저장 안 함":`테스트 저장 완료 ${new Date().toLocaleTimeString("ko-KR",{hour:"2-digit",minute:"2-digit",second:"2-digit"})}`;
      if(submit){activeAttempt={...activeAttempt,status:"SUBMITTED",submittedAt:new Date(),answers:{...assessmentAnswers}};renderAssessmentPortal();}
      return;
    }
    if(!fb)return;
    const ref=fb.fsMod.doc(fb.db,"publicCourses",projectId,"assessments",selectedAssessmentId,"attempts",activeAttempt.studentId),payload={studentId:activeAttempt.studentId,studentName:activeAttempt.studentName||"",authUid:fb.auth.currentUser.uid,status:submit?"SUBMITTED":"IN_PROGRESS",answers:assessmentAnswers,currentQuestion:1,lastSeenAt:fb.fsMod.serverTimestamp(),lastSavedAt:fb.fsMod.serverTimestamp()};if(submit)payload.submittedAt=fb.fsMod.serverTimestamp();
    await fb.fsMod.setDoc(ref,payload,{merge:true});assessmentDirty=false;if(st)st.textContent=submit?"최종 제출됨":`저장 완료 ${new Date().toLocaleTimeString("ko-KR",{hour:"2-digit",minute:"2-digit",second:"2-digit"})}`;if(submit){activeAttempt={...activeAttempt,...payload,status:"SUBMITTED"};localStorage.removeItem(assessmentBackupKey(selectedAssessmentId,activeAttempt.studentId));renderAssessmentPortal();}
  }catch(err){console.error(err);if(st)st.textContent="저장 실패 · 인터넷 연결 확인";}finally{assessmentSaving=false;}
}
async function submitAssessment(){
  const required=(assessmentContent?.questions||[]).filter(q=>q.required!==false),missing=required.filter(q=>!String(assessmentAnswers[q.id]||"").trim());if(missing.length&&!confirm(`작성하지 않은 문항이 ${missing.length}개 있습니다. 그래도 최종 제출할까요?`))return;if(!missing.length&&!confirm("최종 제출하면 학생이 다시 수정할 수 없습니다. 제출할까요?"))return;await saveAssessmentAttempt({submit:true});
}
